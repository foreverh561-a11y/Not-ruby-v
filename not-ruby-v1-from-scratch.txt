--[[
    NOT RUBY V1 - CLEAN REBUILD
    Fresh LocalScript implementation built from scratch.

    Intended location in your own Roblox experience:
    StarterPlayer > StarterPlayerScripts > LocalScript

    This build uses normal Roblox APIs only.
]]

(function()
    local S = {
        services = {},
        fn = {},
        state = {},
        ui = {},
        connections = {},
        markers = {players = {}, monsters = {}},
        original = {},
        theme = {},
    }

    --// SERVICES
    S.services.Players = game:GetService("Players")
    S.services.UIS = game:GetService("UserInputService")
    S.services.RunService = game:GetService("RunService")
    S.services.TweenService = game:GetService("TweenService")
    S.services.Lighting = game:GetService("Lighting")
    S.services.Stats = game:GetService("Stats")
    S.services.TeleportService = game:GetService("TeleportService")

    S.player = S.services.Players.LocalPlayer
    S.camera = workspace.CurrentCamera

    --// STATE
    S.state.menuOpen = true
    S.state.dragLocked = false
    S.state.resizeLocked = false
    S.state.animations = true
    S.state.blurEnabled = false
    S.state.flyEnabled = false
    S.state.flyUp = false
    S.state.flyDown = false
    S.state.noclipEnabled = false
    S.state.sprintEnabled = false
    S.state.playerLocator = false
    S.state.monsterLocator = false
    S.state.walkSpeed = 16
    S.state.jumpPower = 50
    S.state.sprintSpeed = 32
    S.state.flySpeed = 60
    S.state.brightness = 50
    S.state.fov = 70
    S.state.trackerRange = 1500
    S.state.uiScale = 100
    S.state.currentTab = "HOME"
    S.state.destroyed = false
    S.state.currentTheme = 1

    S.theme.list = {
        {name = "RUBY", accent = Color3.fromRGB(225, 45, 60)},
        {name = "BLUE", accent = Color3.fromRGB(70, 145, 255)},
        {name = "PURPLE", accent = Color3.fromRGB(155, 90, 255)},
        {name = "EMERALD", accent = Color3.fromRGB(50, 210, 130)},
        {name = "GOLD", accent = Color3.fromRGB(245, 185, 60)},
    }
    S.theme.bg = Color3.fromRGB(8, 10, 15)
    S.theme.panel = Color3.fromRGB(14, 17, 25)
    S.theme.surface = Color3.fromRGB(22, 26, 37)
    S.theme.surface2 = Color3.fromRGB(29, 34, 48)
    S.theme.text = Color3.fromRGB(242, 244, 250)
    S.theme.muted = Color3.fromRGB(152, 160, 181)
    S.theme.good = Color3.fromRGB(90, 230, 140)
    S.theme.warn = Color3.fromRGB(255, 177, 76)
    S.theme.bad = Color3.fromRGB(255, 92, 110)

    S.fn.accent = function()
        return S.theme.list[S.state.currentTheme].accent
    end

    S.fn.connect = function(signal, callback)
        local c = signal:Connect(callback)
        table.insert(S.connections, c)
        return c
    end

    S.fn.tween = function(object, duration, props, style, direction)
        if not object or not object.Parent then
            return
        end
        if not S.state.animations or duration <= 0 then
            for k, v in pairs(props) do
                object[k] = v
            end
            return
        end
        local info = TweenInfo.new(
            duration,
            style or Enum.EasingStyle.Quint,
            direction or Enum.EasingDirection.Out
        )
        S.services.TweenService:Create(object, info, props):Play()
    end

    S.fn.corner = function(object, radius)
        local c = Instance.new("UICorner")
        c.CornerRadius = UDim.new(0, radius or 10)
        c.Parent = object
        return c
    end

    S.fn.stroke = function(object, transparency, thickness)
        local st = Instance.new("UIStroke")
        st.Name = "ThemeStroke"
        st.Color = S.fn.accent()
        st.Transparency = transparency or 0.45
        st.Thickness = thickness or 1
        st.Parent = object
        return st
    end

    S.fn.label = function(parent, text, size, height, color, bold)
        local l = Instance.new("TextLabel")
        l.BackgroundTransparency = 1
        l.Size = UDim2.new(1, 0, 0, height or 22)
        l.Text = text or ""
        l.TextColor3 = color or S.theme.text
        l.TextSize = size or 12
        l.Font = bold and Enum.Font.GothamBold or Enum.Font.Gotham
        l.TextXAlignment = Enum.TextXAlignment.Left
        l.TextWrapped = true
        l.Parent = parent
        return l
    end

    S.fn.button = function(parent, text, height)
        local b = Instance.new("TextButton")
        b.Size = UDim2.new(1, 0, 0, height or 38)
        b.BackgroundColor3 = S.theme.surface2
        b.BorderSizePixel = 0
        b.AutoButtonColor = false
        b.Text = text or "BUTTON"
        b.TextColor3 = S.theme.text
        b.TextSize = 11
        b.Font = Enum.Font.GothamBold
        b.Parent = parent
        S.fn.corner(b, 10)
        S.fn.stroke(b, 0.75, 1)

        local scale = Instance.new("UIScale")
        scale.Scale = 1
        scale.Parent = b

        S.fn.connect(b.InputBegan, function(input)
            if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
                S.fn.tween(scale, 0.08, {Scale = 0.97}, Enum.EasingStyle.Quad)
            end
        end)
        S.fn.connect(b.InputEnded, function(input)
            if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
                S.fn.tween(scale, 0.1, {Scale = 1}, Enum.EasingStyle.Back)
            end
        end)
        return b
    end

    S.fn.section = function(parent, title, subtitle)
        local card = Instance.new("Frame")
        card.Size = UDim2.new(1, 0, 0, 0)
        card.AutomaticSize = Enum.AutomaticSize.Y
        card.BackgroundColor3 = S.theme.surface
        card.BorderSizePixel = 0
        card.Parent = parent
        S.fn.corner(card, 12)
        S.fn.stroke(card, 0.78, 1)

        local pad = Instance.new("UIPadding")
        pad.PaddingTop = UDim.new(0, 10)
        pad.PaddingBottom = UDim.new(0, 10)
        pad.PaddingLeft = UDim.new(0, 10)
        pad.PaddingRight = UDim.new(0, 10)
        pad.Parent = card

        local layout = Instance.new("UIListLayout")
        layout.Padding = UDim.new(0, 7)
        layout.Parent = card

        local t = S.fn.label(card, title or "SECTION", 12, 18, S.theme.text, true)
        t.Name = "SectionTitle"
        if subtitle and subtitle ~= "" then
            S.fn.label(card, subtitle, 10, 28, S.theme.muted, false)
        end
        return card
    end

    S.fn.row = function(parent)
        local r = Instance.new("Frame")
        r.Size = UDim2.new(1, 0, 0, 38)
        r.BackgroundTransparency = 1
        r.Parent = parent
        local layout = Instance.new("UIListLayout")
        layout.FillDirection = Enum.FillDirection.Horizontal
        layout.Padding = UDim.new(0, 7)
        layout.VerticalAlignment = Enum.VerticalAlignment.Center
        layout.Parent = r
        return r
    end

    S.fn.inputBox = function(parent, placeholder, defaultText)
        local box = Instance.new("TextBox")
        box.Size = UDim2.new(1, 0, 0, 38)
        box.BackgroundColor3 = S.theme.surface2
        box.BorderSizePixel = 0
        box.Text = tostring(defaultText or "")
        box.PlaceholderText = placeholder or ""
        box.ClearTextOnFocus = false
        box.TextColor3 = S.theme.text
        box.PlaceholderColor3 = S.theme.muted
        box.TextSize = 11
        box.Font = Enum.Font.GothamBold
        box.Parent = parent
        S.fn.corner(box, 10)
        S.fn.stroke(box, 0.82, 1)
        return box
    end

    S.fn.setButtonState = function(button, label, enabled)
        if not button then return end
        button.Text = label .. (enabled and " [ON]" or " [OFF]")
        button.TextColor3 = enabled and S.fn.accent() or S.theme.text
    end

    S.fn.status = function(text, mode)
        if not S.ui.status then return end
        S.ui.status.Text = tostring(text or "READY")
        if mode == "bad" then
            S.ui.status.TextColor3 = S.theme.bad
        elseif mode == "warn" then
            S.ui.status.TextColor3 = S.theme.warn
        elseif mode == "good" then
            S.ui.status.TextColor3 = S.theme.good
        else
            S.ui.status.TextColor3 = S.fn.accent()
        end
        S.ui.status.TextTransparency = 0.4
        S.fn.tween(S.ui.status, 0.18, {TextTransparency = 0})
    end

    S.fn.getCharacter = function()
        local ch = S.player.Character
        if not ch then return nil, nil, nil end
        return ch, ch:FindFirstChildOfClass("Humanoid"), ch:FindFirstChild("HumanoidRootPart")
    end

    S.fn.applyMovement = function()
        local _, hum = S.fn.getCharacter()
        if not hum then return end
        hum.WalkSpeed = S.state.sprintEnabled and S.state.sprintSpeed or S.state.walkSpeed
        if hum.UseJumpPower then
            hum.JumpPower = S.state.jumpPower
        end
    end

    S.fn.applyBrightness = function()
        local v = math.clamp(S.state.brightness, 0, 100)
        if v == 50 then
            S.services.Lighting.Brightness = S.original.brightness
            S.services.Lighting.ExposureCompensation = S.original.exposure
            return
        end

        if v < 50 then
            local a = v / 50
            S.services.Lighting.Brightness = S.original.brightness * a
            S.services.Lighting.ExposureCompensation = S.original.exposure - (1.7 * (1 - a))
        else
            local a = (v - 50) / 50
            S.services.Lighting.Brightness = S.original.brightness + (3.5 * a)
            S.services.Lighting.ExposureCompensation = S.original.exposure + (1.6 * a)
        end
    end

    S.fn.updateTheme = function()
        local accent = S.fn.accent()
        if S.ui.headerLine then S.ui.headerLine.BackgroundColor3 = accent end
        if S.ui.status then S.ui.status.TextColor3 = accent end
        if S.ui.openButton then
            local st = S.ui.openButton:FindFirstChild("ThemeStroke")
            if st then st.Color = accent end
        end
        if S.ui.sidebar then S.ui.sidebar.ScrollBarImageColor3 = accent end
        for _, page in pairs(S.ui.pages or {}) do
            page.ScrollBarImageColor3 = accent
        end
        if S.ui.gui then
            for _, d in ipairs(S.ui.gui:GetDescendants()) do
                if d:IsA("UIStroke") and d.Name == "ThemeStroke" then
                    d.Color = accent
                end
                if d:IsA("Frame") and d.Name == "SelectedIndicator" then
                    d.BackgroundColor3 = accent
                end
            end
        end
        if S.ui.tabButtons then
            for name, button in pairs(S.ui.tabButtons) do
                local selected = name == S.state.currentTab
                button.TextColor3 = selected and accent or S.theme.text
                local indicator = button:FindFirstChild("SelectedIndicator")
                if indicator then
                    indicator.BackgroundColor3 = accent
                    indicator.BackgroundTransparency = selected and 0 or 1
                end
            end
        end
    end

    S.fn.makeSlider = function(parent, title, minValue, maxValue, startValue, onChanged)
        local wrap = Instance.new("Frame")
        wrap.Size = UDim2.new(1, 0, 0, 58)
        wrap.BackgroundTransparency = 1
        wrap.Parent = parent

        local titleLabel = Instance.new("TextLabel")
        titleLabel.BackgroundTransparency = 1
        titleLabel.Size = UDim2.new(1, -60, 0, 20)
        titleLabel.Text = title
        titleLabel.TextColor3 = S.theme.text
        titleLabel.TextSize = 10
        titleLabel.Font = Enum.Font.GothamBold
        titleLabel.TextXAlignment = Enum.TextXAlignment.Left
        titleLabel.Parent = wrap

        local valueLabel = Instance.new("TextLabel")
        valueLabel.BackgroundTransparency = 1
        valueLabel.AnchorPoint = Vector2.new(1, 0)
        valueLabel.Position = UDim2.new(1, 0, 0, 0)
        valueLabel.Size = UDim2.fromOffset(55, 20)
        valueLabel.Text = tostring(startValue)
        valueLabel.TextColor3 = S.fn.accent()
        valueLabel.TextSize = 10
        valueLabel.Font = Enum.Font.GothamBold
        valueLabel.TextXAlignment = Enum.TextXAlignment.Right
        valueLabel.Parent = wrap

        local bar = Instance.new("Frame")
        bar.Position = UDim2.new(0, 0, 0, 30)
        bar.Size = UDim2.new(1, 0, 0, 8)
        bar.BackgroundColor3 = S.theme.surface2
        bar.BorderSizePixel = 0
        bar.Active = true
        bar.Parent = wrap
        S.fn.corner(bar, 999)

        local fill = Instance.new("Frame")
        fill.Size = UDim2.new((startValue - minValue) / (maxValue - minValue), 0, 1, 0)
        fill.BackgroundColor3 = S.fn.accent()
        fill.BorderSizePixel = 0
        fill.Parent = bar
        S.fn.corner(fill, 999)

        local knob = Instance.new("Frame")
        knob.AnchorPoint = Vector2.new(0.5, 0.5)
        knob.Position = UDim2.new(fill.Size.X.Scale, 0, 0.5, 0)
        knob.Size = UDim2.fromOffset(18, 18)
        knob.BackgroundColor3 = S.theme.text
        knob.BorderSizePixel = 0
        knob.Parent = bar
        S.fn.corner(knob, 999)
        S.fn.stroke(knob, 0.15, 2)

        local dragging = false
        local function setFromX(x)
            local absoluteX = bar.AbsolutePosition.X
            local width = math.max(bar.AbsoluteSize.X, 1)
            local alpha = math.clamp((x - absoluteX) / width, 0, 1)
            local value = math.floor((minValue + ((maxValue - minValue) * alpha)) + 0.5)
            local exactAlpha = (value - minValue) / (maxValue - minValue)
            fill.Size = UDim2.new(exactAlpha, 0, 1, 0)
            knob.Position = UDim2.new(exactAlpha, 0, 0.5, 0)
            valueLabel.Text = tostring(value)
            fill.BackgroundColor3 = S.fn.accent()
            local st = knob:FindFirstChild("ThemeStroke")
            if st then st.Color = S.fn.accent() end
            onChanged(value)
        end

        S.fn.connect(bar.InputBegan, function(input)
            if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
                dragging = true
                setFromX(input.Position.X)
            end
        end)

        S.fn.connect(S.services.UIS.InputChanged, function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
                setFromX(input.Position.X)
            end
        end)

        S.fn.connect(S.services.UIS.InputEnded, function(input)
            if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
                dragging = false
            end
        end)

        return wrap
    end

    S.fn.makePage = function(name)
        local page = Instance.new("ScrollingFrame")
        page.Name = name .. "Page"
        page.Size = UDim2.fromScale(1, 1)
        page.BackgroundTransparency = 1
        page.BorderSizePixel = 0
        page.ScrollBarThickness = 3
        page.ScrollBarImageColor3 = S.fn.accent()
        page.AutomaticCanvasSize = Enum.AutomaticSize.Y
        page.CanvasSize = UDim2.new()
        page.Visible = false
        page.Parent = S.ui.content

        local pad = Instance.new("UIPadding")
        pad.PaddingTop = UDim.new(0, 10)
        pad.PaddingBottom = UDim.new(0, 12)
        pad.PaddingLeft = UDim.new(0, 10)
        pad.PaddingRight = UDim.new(0, 10)
        pad.Parent = page

        local layout = Instance.new("UIListLayout")
        layout.Padding = UDim.new(0, 8)
        layout.Parent = page
        S.ui.pages[name] = page
        return page
    end

    S.fn.selectTab = function(name)
        if not S.ui.pages[name] then return end
        S.state.currentTab = name
        for pageName, page in pairs(S.ui.pages) do
            page.Visible = pageName == name
        end
        for tabName, button in pairs(S.ui.tabButtons) do
            local selected = tabName == name
            button.BackgroundColor3 = selected and S.theme.surface2 or S.theme.surface
            button.TextColor3 = selected and S.fn.accent() or S.theme.text
            local indicator = button:FindFirstChild("SelectedIndicator")
            if indicator then
                indicator.BackgroundColor3 = S.fn.accent()
                indicator.BackgroundTransparency = selected and 0 or 1
                indicator.Size = UDim2.fromOffset(3, selected and 22 or 0)
            end
        end
        S.fn.status(name)
    end

    S.fn.makeTab = function(name, text)
        local b = Instance.new("TextButton")
        b.Name = name .. "Tab"
        b.Size = UDim2.new(1, 0, 0, 40)
        b.BackgroundColor3 = S.theme.surface
        b.BorderSizePixel = 0
        b.AutoButtonColor = false
        b.Text = text
        b.TextColor3 = S.theme.text
        b.TextSize = 10
        b.Font = Enum.Font.GothamBold
        b.Parent = S.ui.sidebar
        S.fn.corner(b, 9)

        local indicator = Instance.new("Frame")
        indicator.Name = "SelectedIndicator"
        indicator.AnchorPoint = Vector2.new(0, 0.5)
        indicator.Position = UDim2.new(0, 4, 0.5, 0)
        indicator.Size = UDim2.fromOffset(3, 0)
        indicator.BackgroundColor3 = S.fn.accent()
        indicator.BackgroundTransparency = 1
        indicator.BorderSizePixel = 0
        indicator.Parent = b
        S.fn.corner(indicator, 999)

        S.ui.tabButtons[name] = b
        S.fn.connect(b.Activated, function()
            S.fn.selectTab(name)
        end)
        return b
    end

    S.fn.fitPanel = function()
        if not S.camera or not S.ui.panel then return end
        local vp = S.camera.ViewportSize
        local width = math.clamp(math.floor(vp.X * 0.88), 360, 720)
        local height = math.clamp(math.floor(vp.Y * 0.76), 310, 560)
        S.ui.panel.Size = UDim2.fromOffset(width, height)
        S.ui.panel.Position = UDim2.new(0.5, -math.floor(width / 2), 0.5, -math.floor(height / 2))
    end

    S.fn.setMenuOpen = function(open)
        if S.state.destroyed then return end
        S.state.menuOpen = open
        S.ui.openButton.Visible = not open

        if open then
            S.ui.panel.Visible = true
            S.ui.panelScale.Scale = 0.86
            S.ui.panel.BackgroundTransparency = 0.2
            S.fn.tween(S.ui.panelScale, 0.22, {Scale = 1}, Enum.EasingStyle.Back)
            S.fn.tween(S.ui.panel, 0.18, {BackgroundTransparency = 0})
        else
            if not S.state.animations then
                S.ui.panel.Visible = false
                return
            end
            S.fn.tween(S.ui.panelScale, 0.16, {Scale = 0.88}, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
            S.fn.tween(S.ui.panel, 0.16, {BackgroundTransparency = 0.25}, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
            task.delay(0.17, function()
                if S.ui.panel and S.ui.panel.Parent and not S.state.menuOpen then
                    S.ui.panel.Visible = false
                end
            end)
        end
    end

    S.fn.makeMarker = function(rootPart, title, color)
        local gui = Instance.new("BillboardGui")
        gui.Size = UDim2.fromOffset(190, 44)
        gui.StudsOffset = Vector3.new(0, 3.2, 0)
        gui.AlwaysOnTop = true
        gui.MaxDistance = S.state.trackerRange
        gui.Adornee = rootPart
        gui.Parent = rootPart

        local label = Instance.new("TextLabel")
        label.Size = UDim2.fromScale(1, 1)
        label.BackgroundTransparency = 1
        label.Text = title
        label.TextColor3 = color
        label.TextStrokeTransparency = 0.35
        label.TextSize = 13
        label.Font = Enum.Font.GothamBold
        label.TextWrapped = true
        label.Parent = gui
        return gui, label
    end

    S.fn.clearPlayerMarkers = function()
        for target, data in pairs(S.markers.players) do
            if data.gui then data.gui:Destroy() end
            S.markers.players[target] = nil
        end
    end

    S.fn.clearMonsterMarkers = function()
        for target, data in pairs(S.markers.monsters) do
            if data.gui then data.gui:Destroy() end
            S.markers.monsters[target] = nil
        end
    end

    S.fn.refreshPlayerMarkers = function()
        if not S.state.playerLocator then
            S.fn.clearPlayerMarkers()
            return
        end

        local _, _, myRoot = S.fn.getCharacter()
        for _, p in ipairs(S.services.Players:GetPlayers()) do
            if p ~= S.player then
                local ch = p.Character
                local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
                local hum = ch and ch:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    if not S.markers.players[p] or not S.markers.players[p].gui.Parent then
                        local gui, label = S.fn.makeMarker(hrp, "", Color3.fromRGB(100, 190, 255))
                        S.markers.players[p] = {gui = gui, label = label}
                    end
                    local distance = myRoot and math.floor((myRoot.Position - hrp.Position).Magnitude) or 0
                    S.markers.players[p].gui.MaxDistance = S.state.trackerRange
                    S.markers.players[p].label.Text = p.DisplayName .. " (@" .. p.Name .. ")\n" .. tostring(distance) .. " studs"
                else
                    local old = S.markers.players[p]
                    if old and old.gui then old.gui:Destroy() end
                    S.markers.players[p] = nil
                end
            end
        end

        for p, data in pairs(S.markers.players) do
            if not p.Parent then
                if data.gui then data.gui:Destroy() end
                S.markers.players[p] = nil
            end
        end
    end

    S.fn.isMonster = function(model)
        if not model:IsA("Model") then return false end
        local hum = model:FindFirstChildOfClass("Humanoid")
        local root = model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("UpperTorso") or model:FindFirstChild("Torso")
        if not hum or not root or hum.Health <= 0 then return false end
        for _, p in ipairs(S.services.Players:GetPlayers()) do
            if p.Character == model then return false end
        end
        return true
    end

    S.fn.refreshMonsterMarkers = function()
        if not S.state.monsterLocator then
            S.fn.clearMonsterMarkers()
            return
        end

        local _, _, myRoot = S.fn.getCharacter()
        local seen = {}
        for _, obj in ipairs(workspace:GetChildren()) do
            if S.fn.isMonster(obj) then
                seen[obj] = true
                local root = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("UpperTorso") or obj:FindFirstChild("Torso")
                local distance = myRoot and math.floor((myRoot.Position - root.Position).Magnitude) or 0
                if distance <= S.state.trackerRange then
                    if not S.markers.monsters[obj] or not S.markers.monsters[obj].gui.Parent then
                        local gui, label = S.fn.makeMarker(root, "", Color3.fromRGB(255, 190, 70))
                        S.markers.monsters[obj] = {gui = gui, label = label}
                    end
                    S.markers.monsters[obj].gui.MaxDistance = S.state.trackerRange
                    S.markers.monsters[obj].label.Text = obj.Name .. "\n" .. tostring(distance) .. " studs"
                else
                    local old = S.markers.monsters[obj]
                    if old and old.gui then old.gui:Destroy() end
                    S.markers.monsters[obj] = nil
                end
            end
        end

        for model, data in pairs(S.markers.monsters) do
            if not seen[model] or not model.Parent then
                if data.gui then data.gui:Destroy() end
                S.markers.monsters[model] = nil
            end
        end
    end

    S.fn.enableNoclip = function(enabled)
        S.state.noclipEnabled = enabled
        if not enabled then
            local ch = S.player.Character
            if ch then
                for _, part in ipairs(ch:GetDescendants()) do
                    if part:IsA("BasePart") and S.original.collisions[part] ~= nil then
                        part.CanCollide = S.original.collisions[part]
                    end
                end
            end
            S.original.collisions = {}
        end
    end

    S.fn.destroy = function()
        if S.state.destroyed then return end
        S.state.destroyed = true
        S.state.flyEnabled = false
        S.state.noclipEnabled = false
        S.fn.clearPlayerMarkers()
        S.fn.clearMonsterMarkers()
        S.fn.enableNoclip(false)

            if S.state.dragLocked then return end

                startPos = input.Position
                startPanelPos = S.ui.panel.Position
            end
        end)

        S.fn.connect(S.services.UIS.InputChanged, function(input)
            if not dragging or S.state.dragLocked then return end
            if input == dragInput or input.UserInputType == Enum.UserInputType.MouseMovement then
                local delta = input.Position - startPos
                S.ui.panel.Position = UDim2.new(
                    startPanelPos.X.Scale,
                    startPanelPos.X.Offset + delta.X,
                    startPanelPos.Y.Scale,
                    startPanelPos.Y.Offset + delta.Y

                dragInput = nil
            end

        if not ch or not hum or not root then return end

        if S.state.noclipEnabled then
            for _, part in ipairs(ch:GetDescendants()) do
                if part:IsA("BasePart") then
                    if S.original.collisions[part] == nil then
                        S.original.collisions[part] = part.CanCollide
                    end
                    part.CanCollide = false
                end
            end
        end

        if S.state.flyEnabled then
            hum.AutoRotate = false
            local move = hum.MoveDirection
            local vertical = 0
            if S.state.flyUp then vertical += 1 end
            if S.state.flyDown then vertical -= 1 end
            local velocity = (move * S.state.flySpeed) + Vector3.new(0, vertical * S.state.flySpeed, 0)
            root.AssemblyLinearVelocity = velocity
            if move.Magnitude > 0.05 then
                local look = Vector3.new(move.X, 0, move.Z)
                if look.Magnitude > 0.05 then
                    root.CFrame = CFrame.lookAt(root.Position, root.Position + look)
                end
            end
        else
            hum.AutoRotate = true
        end
    end)
